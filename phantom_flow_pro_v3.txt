// 🧠 REGIME ENGINE — Math
// ============================================================
logRet = math.log(close / nz(close[1], close))

// 1. Trend Efficiency Ratio (Kaufman)
netMove    = math.abs(close - nz(close[regimeWindow], close))
pathLen    = math.sum(math.abs(ta.change(close)), regimeWindow)
efficiency = pathLen == 0 ? 0.0 : netMove / pathLen

// 2. Lagged Autocorrelation — Hurst proxy across 3 lags
ac1         = nz(ta.correlation(logRet, logRet[1], regimeWindow), 0.0)
ac2         = nz(ta.correlation(logRet, logRet[2], regimeWindow), 0.0)
ac3         = nz(ta.correlation(logRet, logRet[3], regimeWindow), 0.0)
persistence = math.avg(ac1, ac2, ac3)

// 3. Volatility-of-Volatility Z-Score
realizedVol = ta.stdev(logRet, 20)
vovRaw      = ta.stdev(realizedVol, 50)
vovMean     = ta.sma(vovRaw, 200)
vovStd      = ta.stdev(vovRaw, 200)
vovZ        = vovStd > 0 ? (vovRaw - vovMean) / vovStd : 0.0

// 4. Classification
warmupBars  = 200
regimeReady = bar_index > warmupBars and not na(efficiency) and not na(persistence)

regime = not regimeReady ? 'INIT' :
         vovZ > breakoutZ ? 'BREAKOUT' :
         efficiency < deadThresh ? 'DEAD' :
         (efficiency > trendThresh and persistence > persistMin) ? 'TRENDING' :
         'RANGING'

// 5. Confidence 0-100
regimeConf = regime == 'TRENDING' ? math.min(100.0, math.max(0.0, (efficiency - trendThresh) * 250.0 + math.max(0.0, persistence) * 200.0)) :
             regime == 'RANGING'  ? math.min(100.0, math.max(0.0, (trendThresh - efficiency) * 200.0)) :
             regime == 'BREAKOUT' ? math.min(100.0, math.max(0.0, (vovZ - breakoutZ) * 40.0)) :
             regime == 'DEAD'     ? math.min(100.0, math.max(0.0, (deadThresh - efficiency) * 600.0)) :
             0.0

// 6. Direction (for trending/breakout regimes)
shortMA   = ta.ema(close, 21)
longMA    = ta.ema(close, 55)
direction = shortMA > longMA ? 1 : shortMA < longMA ? -1 : 0

// 7. Regime change flag
regimeChanged = regime != regime[1] and regime != 'INIT' and regime[1] != 'INIT'


// ============================================================
// 🧠 REGIME ENGINE — Multi-Timeframe (replaces ta.supertrend MTF)

// Compute the same regime engine on D and W timeframes for confluence.
htfRegimeFn() =>
    htfRet    = math.log(close / nz(close[1], close))
    htfNet    = math.abs(close - nz(close[regimeWindow], close))
    htfPath   = math.sum(math.abs(ta.change(close)), regimeWindow)
    htfEff    = htfPath == 0 ? 0.0 : htfNet / htfPath
    htfAc1    = nz(ta.correlation(htfRet, htfRet[1], regimeWindow), 0.0)
    htfPers   = htfAc1
    htfRegime = htfEff > trendThresh and htfPers > persistMin ? 1 :
                htfEff < deadThresh ? -2 :
                0
    htfShort  = ta.ema(close, 21)
    htfLong   = ta.ema(close, 55)
    htfDir    = htfShort > htfLong ? 1 : htfShort < htfLong ? -1 : 0
    [htfRegime, htfDir]

[htfRegime_D, htfDir_D] = request.security(syminfo.tickerid, 'D', htfRegimeFn(), lookahead = barmerge.lookahead_off)
[htfRegime_W, htfDir_W] = request.security(syminfo.tickerid, 'W', htfRegimeFn(), lookahead = barmerge.lookahead_off)

htfTrendingBull = htfRegime_D == 1 and htfDir_D == 1 and htfDir_W == 1
htfTrendingBear = htfRegime_D == 1 and htfDir_D == -1 and htfDir_W == -1
htfAligned      = htfDir_D == htfDir_W and htfDir_D != 0
